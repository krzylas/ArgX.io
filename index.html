-- ===================================================== -- FIX STORAGE PERMISSIONS - SUPABASE -- This fixes the "Permission denied" error for uploads -- ===================================================== -- ===================================================== -- STEP 1: DELETE OLD CONFLICTING POLICIES -- ===================================================== -- Remove any existing policies that might conflict DROP POLICY IF EXISTS "Allow public uploads" ON storage.objects; DROP POLICY IF EXISTS "Allow public access to images" ON storage.objects; DROP POLICY IF EXISTS "Allow public delete" ON storage.objects; DROP POLICY IF EXISTS "Public Access" ON storage.objects; DROP POLICY IF EXISTS "Public Upload" ON storage.objects; -- ===================================================== -- STEP 2: CREATE NEW PERMISSIVE POLICIES -- ===================================================== -- Policy 1: Allow ANYONE (including anonymous) to INSERT/UPLOAD CREATE POLICY "Enable insert for all users" ON storage.objects FOR INSERT TO public WITH CHECK (bucket_id = 'product-images'); -- Policy 2: Allow ANYONE to SELECT/VIEW images CREATE POLICY "Enable select for all users" ON storage.objects FOR SELECT TO public USING (bucket_id = 'product-images'); -- Policy 3: Allow ANYONE to UPDATE images CREATE POLICY "Enable update for all users" ON storage.objects FOR UPDATE TO public USING (bucket_id = 'product-images') WITH CHECK (bucket_id = 'product-images'); -- Policy 4: Allow ANYONE to DELETE images CREATE POLICY "Enable delete for all users" ON storage.objects FOR DELETE TO public USING (bucket_id = 'product-images'); -- ===================================================== -- STEP 3: VERIFY BUCKET EXISTS AND IS PUBLIC -- ===================================================== -- Check if bucket exists DO $$ BEGIN -- Try to update bucket to be public UPDATE storage.buckets SET public = true WHERE id = 'product-images'; -- If no rows were updated, bucket doesn't exist IF NOT FOUND THEN -- Create the bucket INSERT INTO storage.buckets (id, name, public) VALUES ('product-images', 'product-images', true); RAISE NOTICE 'Created new bucket: product-images'; ELSE RAISE NOTICE 'Updated existing bucket to public'; END IF; END $$; -- ===================================================== -- STEP 4: SET BUCKET FILE SIZE AND MIME TYPES -- ===================================================== -- Update bucket settings UPDATE storage.buckets SET file_size_limit = 5242880, -- 5MB limit allowed_mime_types = ARRAY['image/jpeg', 'image/jpg', 'image/png', 'image/webp'] WHERE id = 'product-images'; -- ===================================================== -- STEP 5: VERIFY POLICIES ARE ACTIVE -- ===================================================== -- List all policies for product-images bucket SELECT schemaname, tablename, policyname, permissive, roles, cmd, qual, with_check FROM pg_policies WHERE tablename = 'objects' AND policyname LIKE '%Enable%' ORDER BY policyname; -- ===================================================== -- STEP 6: TEST THE PERMISSIONS -- ===================================================== -- This query should return true if policies are working SELECT EXISTS ( SELECT 1 FROM storage.buckets WHERE id = 'product-images' AND public = true ) as bucket_is_public;
